```ts
/**
 * Text as the studio page shows it (TAT-48). Pure, so the loaders, the head
 * and the tests share it. Everything the API sends is plain text, rendered
 * as React text, never as HTML.
 */

/** Control characters other than tab and newline. */
const CONTROL = /(?![\t\n])\p{Cc}/gu;

/**
 * Multi-line text from the API (an intro, a bio, an address, a closed
 * message), ready for `white-space: pre-line`: Windows and old Mac line ends
 * become "\n", control characters go, spaces before a line break go, more
 * than one blank line becomes one, and the ends are trimmed. Joiners
 * (U+200C, U+200D) stay: Devanagari and emoji need them. Blank text is null.
 */
export function cleanText(value: string | null | undefined): string | null {
	if (typeof value !== "string") return null;
	const text = value
		.replace(/\r\n?/g, "\n")
		.replace(CONTROL, "")
		.replace(/[ \t]+\n/g, "\n")
		.replace(/\n{3,}/g, "\n\n")
		.trim();
	return text === "" ? null : text;
}

/** One-line text (a name): every run of whitespace becomes one space. */
export function cleanLine(value: string | null | undefined): string | null {
	const text = cleanText(value);
	return text === null ? null : text.replace(/\s+/gu, " ");
}

/** The longest meta description, in user-perceived characters. */
export const DESCRIPTION_MAX = 160;

/**
 * A meta description from free text: one line, at most `max` graphemes
 * (user-perceived characters, so a Devanagari cluster or an emoji is never
 * split), cut at the last space when one falls in the second half, and
 * ending in "…" when cut.
 */
export function trimDescription(text: string, max = DESCRIPTION_MAX): string {
	const flat = text.replace(/\s+/gu, " ").trim();
	const parts = graphemes(flat);
	if (parts.length <= max) return flat;
	const kept = parts.slice(0, max - 1);
	const space = kept.lastIndexOf(" ");
	const cut = space >= Math.floor(max / 2) ? kept.slice(0, space) : kept;
	return `${cut.join("").trimEnd()}…`;
}

function graphemes(text: string): string[] {
	if (typeof Intl.Segmenter === "function") {
		const segmenter = new Intl.Segmenter(undefined, {
			granularity: "grapheme",
		});
		return Array.from(segmenter.segment(text), (part) => part.segment);
	}
	return Array.from(text);
}
```

```ts
/**
 * The links the studio page builds (TAT-48). Pure. Only these ever reach an
 * href: the page's own paths and fragments, https: (from the API, parsed
 * again here), tel:, mailto: and viber://chat. The chat links copy the
 * platform's chatLinks() (apps/platform/app/modules/clients/contact_links.ts),
 * which the storefront may not import.
 */

/** E.164, as the platform stores contact phones (TAT-26). */
const E164 = /^\+[1-9]\d{6,14}$/;

/**
 * A conservative email shape: no spaces, quotes, brackets, "?", "#", "&",
 * "%", ":" or "/" (nothing that could add a header or change the scheme),
 * and a dotted domain.
 */
const EMAIL = /^[^\s@"<>()[\]\\,;:?#&%/]+@[A-Za-z0-9-]+(?:\.[A-Za-z0-9-]+)+$/;

const MAX_EMAIL_LENGTH = 254;
const MAX_URL_LENGTH = 2048;

/** What the page may put in an href. */
const ALLOWED_HREF =
	/^(?:\/(?![/\\])|\/$|#|https:\/\/|tel:\+|mailto:|viber:\/\/chat\?)/;

export function isPhone(value: unknown): value is string {
	return typeof value === "string" && E164.test(value);
}

export function isEmail(value: unknown): value is string {
	return (
		typeof value === "string" &&
		value.length <= MAX_EMAIL_LENGTH &&
		EMAIL.test(value)
	);
}

/** A WhatsApp chat with `phone`; `text` prefills the message. */
export function whatsappLink(
	phone: string | null | undefined,
	text?: string,
): string | null {
	if (!isPhone(phone)) return null;
	const digits = phone.slice(1);
	return text === undefined
		? `https://wa.me/${digits}`
		: `https://wa.me/${digits}?text=${encodeURIComponent(text)}`;
}

/** A Viber chat with `phone` (Viber's link takes no message). */
export function viberLink(phone: string | null | undefined): string | null {
	return isPhone(phone) ? `viber://chat?number=%2B${phone.slice(1)}` : null;
}

/** A call to `phone`. */
export function telLink(phone: string | null | undefined): string | null {
	return isPhone(phone) ? `tel:+${phone.slice(1)}` : null;
}

/** An email to `email`, with an optional prefilled subject and body. */
export function mailtoLink(
	email: string | null | undefined,
	message?: { subject: string; body: string },
): string | null {
	if (!isEmail(email)) return null;
	if (!message) return `mailto:${email}`;
	return `mailto:${email}?subject=${encodeURIComponent(message.subject)}&body=${encodeURIComponent(message.body)}`;
}

/** A Google Maps search for `query` (no key, no embed). */
export function mapsLink(query: string): string {
	return `https://www.google.com/maps/search/?api=1&query=${encodeURIComponent(query)}`;
}

/** An https URL without credentials, normalized, or null. */
export function safeHttpsUrl(value: unknown): string | null {
	const url = parseUrl(value);
	return url?.protocol === "https:" ? url.href : null;
}

/** An http(s) image URL without credentials, normalized, or null. */
export function safeImageUrl(value: unknown): string | null {
	const url = parseUrl(value);
	return url && (url.protocol === "https:" || url.protocol === "http:")
		? url.href
		: null;
}

/** The host a link goes to, without "www.": the label of an unknown platform. */
export function linkHostname(href: string): string {
	return new URL(href).hostname.replace(/^www\./, "");
}

/** Web links never tell the next site which page sent the visitor. */
export function linkRel(href: string): string | undefined {
	return /^https?:\/\//i.test(href) ? "noopener noreferrer" : undefined;
}

/** Whether the page may render `href` (the components check every link). */
export function isAllowedHref(href: string): boolean {
	return ALLOWED_HREF.test(href);
}

function parseUrl(value: unknown): URL | undefined {
	if (
		typeof value !== "string" ||
		value.length > MAX_URL_LENGTH ||
		value !== value.trim()
	) {
		return undefined;
	}
	try {
		const url = new URL(value);
		return url.username || url.password || !url.hostname ? undefined : url;
	} catch {
		return undefined;
	}
}
```

```ts
import type { Artist, Studio } from "@tattoo-drip/sdk";
import {
	isEmail,
	isPhone,
	linkHostname,
	mailtoLink,
	mapsLink,
	safeHttpsUrl,
	safeImageUrl,
	telLink,
	viberLink,
	whatsappLink,
} from "./links.ts";
import { cleanLine, cleanText } from "./text.ts";

/**
 * What the studio page shows (TAT-48), built from TAT-44's allow-list only.
 * Every link is built or checked here; components render these as they are.
 */

/**
 * The shape of an artist slug: a copy of the contract's `ArtistSlug` pattern
 * (packages/types/openapi.yaml), pinned by test/studio/view-model.test.ts.
 */
export const ARTIST_SLUG = /^[a-z0-9][a-z0-9-]{0,61}[a-z0-9]$/;

export type ContactLinkKind =
	| "instagram"
	| "whatsapp"
	| "viber"
	| "call"
	| "email"
	| "facebook"
	| "tiktok"
	| "website"
	| "other";

export interface ContactLink {
	readonly kind: ContactLinkKind;
	readonly label: string;
	readonly href: string;
}

export interface StudioView {
	readonly slug: string;
	readonly name: string;
	readonly type: Studio["type"];
	readonly typeLabel: string;
	readonly intro: string | null;
	readonly accepting: boolean;
	/** E.164, checked. */
	readonly phone: string | null;
	readonly email: string | null;
	readonly address: { readonly text: string; readonly mapsHref: string } | null;
	/** Instagram first, then the ways to reach the studio, then other places. */
	readonly links: readonly ContactLink[];
	/** The social links' https URLs, in the API's order (JSON-LD `sameAs`). */
	readonly socialUrls: readonly string[];
}

export interface ArtistPhoto {
	readonly src: string;
	readonly width: number;
	readonly height: number;
	readonly alt: string;
}

export interface ArtistView {
	readonly id: string;
	readonly slug: string;
	readonly name: string;
	readonly bio: string | null;
	readonly photo: ArtistPhoto | null;
	readonly accepting: boolean;
	/** Only while the artist isn't accepting inquiries. */
	readonly closedMessage: string | null;
	readonly pageHref: string;
	readonly inquiryHref: string;
}

const TYPE_LABELS: Record<Studio["type"], string> = {
	studio: "Tattoo studio",
	independent: "Independent tattoo artist",
};

const SOCIAL: Record<string, { kind: ContactLinkKind; label: string }> = {
	instagram: { kind: "instagram", label: "Instagram" },
	facebook: { kind: "facebook", label: "Facebook" },
	tiktok: { kind: "tiktok", label: "TikTok" },
	website: { kind: "website", label: "Website" },
};

const LINK_ORDER: readonly ContactLinkKind[] = [
	"instagram",
	"whatsapp",
	"viber",
	"call",
	"email",
	"facebook",
	"tiktok",
	"website",
	"other",
];

/** Photos are at most 2048 px (TAT-44); anything outside this is no photo. */
const MAX_PHOTO_SIDE = 10_000;

export function studioViewModel(studio: Studio): StudioView {
	const name = cleanLine(studio.name) ?? studio.slug;
	const phone = isPhone(studio.contact.phone) ? studio.contact.phone : null;
	const email = isEmail(studio.contact.email) ? studio.contact.email : null;
	const addressText = cleanText(studio.contact.address);

	const social: ContactLink[] = [];
	for (const link of studio.socialLinks) {
		const href = safeHttpsUrl(link.url);
		if (!href) continue;
		const known = Object.hasOwn(SOCIAL, link.platform)
			? SOCIAL[link.platform]
			: undefined;
		social.push({
			kind: known?.kind ?? "other",
			label: known?.label ?? linkHostname(href),
			href,
		});
	}

	const contact: ContactLink[] = [];
	const whatsapp = whatsappLink(phone);
	if (whatsapp) contact.push({ kind: "whatsapp", label: "WhatsApp", href: whatsapp });
	const viber = viberLink(phone);
	if (viber) contact.push({ kind: "viber", label: "Viber", href: viber });
	const call = telLink(phone);
	if (call) contact.push({ kind: "call", label: "Call", href: call });
	const mail = mailtoLink(email);
	if (mail) contact.push({ kind: "email", label: "Email", href: mail });

	// Array.prototype.sort is stable: links of one kind keep the API's order.
	const links = [...social, ...contact].sort(
		(a, b) => LINK_ORDER.indexOf(a.kind) - LINK_ORDER.indexOf(b.kind),
	);

	return {
		slug: studio.slug,
		name,
		type: studio.type,
		typeLabel: TYPE_LABELS[studio.type] ?? TYPE_LABELS.studio,
		intro: cleanText(studio.intro),
		accepting: studio.acceptingInquiries === true,
		phone,
		email,
		address: addressText
			? {
					text: addressText,
					mapsHref: mapsLink(`${name}, ${addressText.replace(/\n+/g, ", ")}`),
				}
			: null,
		links,
		socialUrls: social.map((link) => link.href),
	};
}

/** The visible artists as the page shows them; one whose slug can't be a link is left out. */
export function artistViewModels(artists: readonly Artist[]): ArtistView[] {
	return artists
		.filter((artist) => ARTIST_SLUG.test(artist.slug))
		.map(artistViewModel);
}

/** The artist with exactly this slug, or null. */
export function findArtist(
	artists: readonly ArtistView[],
	slug: string,
): ArtistView | null {
	if (!ARTIST_SLUG.test(slug)) return null;
	return artists.find((artist) => artist.slug === slug) ?? null;
}

function artistViewModel(artist: Artist): ArtistView {
	const name = cleanLine(artist.name) ?? artist.slug;
	const accepting = artist.acceptingInquiries === true;
	return {
		id: artist.id,
		slug: artist.slug,
		name,
		bio: cleanText(artist.bio),
		photo: photoView(artist.photo, name),
		accepting,
		closedMessage: accepting ? null : cleanText(artist.closedMessage),
		pageHref: `/artists/${artist.slug}`,
		inquiryHref: `/artists/${artist.slug}#inquiry`,
	};
}

function photoView(photo: Artist["photo"], name: string): ArtistPhoto | null {
	if (!photo) return null;
	const src = safeImageUrl(photo.url);
	if (!src || !isSide(photo.width) || !isSide(photo.height)) return null;
	return { src, width: photo.width, height: photo.height, alt: `Photo of ${name}` };
}

function isSide(value: unknown): value is number {
	return Number.isInteger(value) && (value as number) >= 1 && (value as number) <= MAX_PHOTO_SIDE;
}
```

```ts
import { mailtoLink, whatsappLink } from "./links.ts";
import type { ArtistView, StudioView } from "./view-model.ts";

/**
 * The #inquiry section's state (TAT-48): the contract TAT-57's intake form
 * fills. `open` is where the form goes: it sends `preferredArtist.id` as
 * createInquiry's `preferredArtistId` and lets the client choose only among
 * `artists` (the visible ones taking inquiries, by name).
 */

/** An artist an inquiry can ask for. */
export interface InquiryArtist {
	readonly id: string;
	readonly name: string;
}

export type InquiryState =
	| {
			readonly kind: "open";
			readonly studioName: string;
			readonly preferredArtist: InquiryArtist | null;
			readonly artists: readonly InquiryArtist[];
	  }
	| { readonly kind: "studio-closed"; readonly studioName: string }
	| {
			readonly kind: "artist-closed";
			readonly studioName: string;
			readonly artistName: string;
			/** Whether a general inquiry (no artist) would be taken. */
			readonly studioOpen: boolean;
	  };

/** How an open inquiry is sent until TAT-57's form exists. */
export interface InquiryChannels {
	/** WhatsApp with a prefilled message. */
	readonly whatsapp: string | null;
	/** An email with a prefilled subject and body. */
	readonly email: string | null;
	/** The studio's Instagram, only when it has neither phone nor email. */
	readonly instagram: string | null;
}

/**
 * The studio page (`selected` null) goes by the studio's flag; an artist's
 * page by that artist's own flag, which is what intake checks for an
 * inquiry naming them.
 */
export function inquiryState(
	studio: StudioView,
	artists: readonly ArtistView[],
	selected: ArtistView | null,
): InquiryState {
	const open = artists
		.filter((artist) => artist.accepting)
		.map(({ id, name }) => ({ id, name }));
	if (selected) {
		return selected.accepting
			? {
					kind: "open",
					studioName: studio.name,
					preferredArtist: { id: selected.id, name: selected.name },
					artists: open,
				}
			: {
					kind: "artist-closed",
					studioName: studio.name,
					artistName: selected.name,
					studioOpen: studio.accepting,
				};
	}
	return studio.accepting
		? { kind: "open", studioName: studio.name, preferredArtist: null, artists: open }
		: { kind: "studio-closed", studioName: studio.name };
}

export function inquiryChannels(
	studio: StudioView,
	artist: InquiryArtist | null,
): InquiryChannels {
	const ask = artist ? `ask ${artist.name}` : "ask";
	const whatsapp = whatsappLink(
		studio.phone,
		`Hi ${studio.name}, I'd like to ${ask} about a tattoo.`,
	);
	const email = mailtoLink(studio.email, {
		subject: artist ? `Tattoo inquiry for ${artist.name}` : "Tattoo inquiry",
		body: `Hi ${studio.name},\r\n\r\nMy idea:\r\nPlacement and size:\r\nPreferred dates:\r\n`,
	});
	const instagram =
		whatsapp || email
			? null
			: (studio.links.find((link) => link.kind === "instagram")?.href ?? null);
	return { whatsapp, email, instagram };
}
```

```ts
import { trimDescription } from "./text.ts";
import type { ArtistPhoto, ArtistView, StudioView } from "./view-model.ts";

/**
 * Each page's head (TAT-48): title, description, canonical URL, Open Graph
 * and, on the studio page, minimal JSON-LD of what the page shows.
 */

export interface PageSeo {
	readonly title: string;
	readonly description: string;
	/** Absolute, from the site origin (src/site/). */
	readonly canonical: string;
	readonly type: "website" | "profile";
	readonly siteName: string;
	readonly image: ArtistPhoto | null;
	readonly jsonLd: Record<string, unknown> | null;
}

export type HeadMeta =
	| { title: string }
	| { name: string; content: string }
	| { property: string; content: string };

/** The shape a route's `head` returns. */
export interface HeadTags {
	meta: HeadMeta[];
	links: { rel: string; href: string }[];
	scripts: { type: string; children: string }[];
}

export function studioPageSeo(
	studio: StudioView,
	artists: readonly ArtistView[],
	origin: string,
): PageSeo {
	const canonical = `${origin}/`;
	const description = studio.intro
		? trimDescription(studio.intro)
		: studio.type === "independent"
			? `${studio.name}, independent tattoo artist. Send an inquiry.`
			: `${studio.name}, tattoo studio. See the artists and send an inquiry.`;
	// A studio's only artist is the studio's face; among several, no one is.
	const image = artists.length === 1 ? artists[0].photo : null;
	return {
		title: studio.name,
		description,
		canonical,
		type: "website",
		siteName: studio.name,
		image,
		jsonLd: studioJsonLd(studio, canonical, description, image),
	};
}

export function artistPageSeo(
	studio: StudioView,
	artist: ArtistView,
	origin: string,
): PageSeo {
	const sameName = artist.name === studio.name;
	const description = artist.bio
		? trimDescription(artist.bio)
		: sameName
			? `${artist.name} is a tattoo artist. Send an inquiry.`
			: `${artist.name} is a tattoo artist at ${studio.name}. Send an inquiry.`;
	return {
		title: sameName ? artist.name : `${artist.name} at ${studio.name}`,
		description,
		canonical: `${origin}${artist.pageHref}`,
		type: "profile",
		siteName: studio.name,
		image: artist.photo,
		jsonLd: null,
	};
}

/** schema.org TattooParlor with only the public fields the page shows. */
export function studioJsonLd(
	studio: StudioView,
	canonical: string,
	description: string,
	image: ArtistPhoto | null,
): Record<string, unknown> {
	return {
		"@context": "https://schema.org",
		"@type": "TattooParlor",
		name: studio.name,
		url: canonical,
		description,
		...(studio.phone ? { telephone: studio.phone } : {}),
		...(studio.email ? { email: studio.email } : {}),
		...(studio.address
			? { address: studio.address.text.replace(/\n+/g, ", ") }
			: {}),
		...(studio.socialUrls.length > 0 ? { sameAs: [...studio.socialUrls] } : {}),
		...(image ? { image: image.src } : {}),
	};
}

const SCRIPT_ESCAPES: Record<string, string> = {
	"<": "\\u003c",
	">": "\\u003e",
	"&": "\\u0026",
	" ": "\\u2028",
	" ": "\\u2029",
};

/**
 * JSON for a `<script type="application/ld+json">`. The router inserts head
 * script children as raw HTML, so nothing here may close the element: the
 * characters that could are written as JSON escapes (still the same JSON).
 */
export function serializeJsonLd(value: unknown): string {
	return JSON.stringify(value).replace(
		/[<>&  ]/g,
		(character) => SCRIPT_ESCAPES[character] ?? character,
	);
}

export function headTags(seo: PageSeo): HeadTags {
	const meta: HeadMeta[] = [
		{ title: seo.title },
		{ name: "description", content: seo.description },
		{ property: "og:type", content: seo.type },
		{ property: "og:site_name", content: seo.siteName },
		{ property: "og:title", content: seo.title },
		{ property: "og:description", content: seo.description },
		{ property: "og:url", content: seo.canonical },
	];
```

```ts
/**
 * The studio page's own origin, e.g. https://blackneedle.tattoodrip.com
 * (TAT-48): canonical links and Open Graph URLs must be absolute. Set once,
 * like TAT-23's TattooClientRef: on the server from STOREFRONT_URL and the
 * slug the tenant middleware confirmed (src/site/request.ts), in the browser
 * from the router's dehydrated state. The browser never reads its own host
 * (TAT-12). The origin is public: it is in every page's canonical link.
 */

const NOT_SET =
	"The site origin is not set yet: getRouter() sets it on the server, the router's hydrate in the browser.";
const INVALID = "This page's site address is missing or invalid. Reload the page.";

/** The router context's part for the site (next to TAT-23's ApiContext). */
export interface SiteContext {
	readonly site: SiteRef;
}

/** What the router's dehydrated state carries for the site. */
export interface DehydratedSite {
	site: { origin: string };
}

export class SiteRef {
	#origin: string | undefined;

	connect(origin: string): void {
		if (this.#origin !== undefined) {
			throw new Error("The site origin is already set.");
		}
		this.#origin = parseSiteOrigin(origin);
	}

	get origin(): string {
		if (this.#origin === undefined) throw new Error(NOT_SET);
		return this.#origin;
	}
}

/** An http(s) origin exactly (no path, no trailing slash, lowercase), or a fixed error. */
export function parseSiteOrigin(value: unknown): string {
	if (typeof value === "string") {
		const url = parseUrl(value);
		if (
			url &&
			(url.protocol === "https:" || url.protocol === "http:") &&
			url.origin === value
		) {
			return value;
		}
	}
	throw new Error(INVALID);
}

/** The router's dehydrate: the origin, and nothing else. */
export function dehydrateSite(site: SiteRef): DehydratedSite {
	return { site: { origin: site.origin } };
}

/** The router's hydrate: sets the browser's origin from the server's. */
export function hydrateSite(
	site: SiteRef,
	dehydrated: { site?: unknown } | undefined,
): void {
	const value = dehydrated?.site;
	const origin =
		typeof value === "object" && value !== null
			? (value as Record<string, unknown>).origin
			: undefined;
	site.connect(parseSiteOrigin(origin));
}

function parseUrl(value: string): URL | undefined {
	try {
		return new URL(value);
	} catch {
		return undefined;
	}
}
```

```ts
/**
 * The studio page's loaders (TAT-48). They read through TAT-23's cache
 * (`query` with staleTime "static": cached data, else one fetch) and return
 * view models, so components need no query hooks and a CDN-aged page never
 * refetches in the browser. Loader data is serialized into the page: only
 * public, already-shown data goes in it.
 */

export type PageContext = ApiContext & SiteContext;

export interface StudioPageData {
	readonly studio: StudioView;
	readonly artists: readonly ArtistView[];
	readonly inquiry: InquiryState;
	readonly channels: InquiryChannels;
	readonly seo: PageSeo;
}

export interface ArtistPageData {
	readonly studio: StudioView;
	readonly artist: ArtistView;
	readonly inquiry: InquiryState;
	readonly channels: InquiryChannels;
	readonly seo: PageSeo;
}

export function studioPageData(
	studio: Studio,
	artists: readonly Artist[],
	origin: string,
): StudioPageData {
	const studioView = studioViewModel(studio);
	const artistViews = artistViewModels(artists);
	const inquiry = inquiryState(studioView, artistViews, null);
	return {
		studio: studioView,
		artists: artistViews,
		inquiry,
		channels: channelsFor(studioView, inquiry),
		seo: studioPageSeo(studioView, artistViews, origin),
	};
}

/** The artist page's data, or null when no visible artist has exactly `slug`. */
export function artistPageData(
	studio: Studio,
	artists: readonly Artist[],
	slug: string,
	origin: string,
): ArtistPageData | null {
	const studioView = studioViewModel(studio);
	const artistViews = artistViewModels(artists);
	const artist = findArtist(artistViews, slug);
	if (!artist) return null;
	const inquiry = inquiryState(studioView, artistViews, artist);
	return {
		studio: studioView,
		artist,
		inquiry,
		channels: channelsFor(studioView, inquiry),
		seo: artistPageSeo(studioView, artist, origin),
	};
}

/** `/`: the studio (already cached by the tenant middleware) and its artists. */
export async function loadStudioPage(context: PageContext): Promise<StudioPageData> {
	const [studio, artists] = await Promise.all([
		loadStudio(context),
		loadArtists(context),
	]);
	return studioPageData(studio, artists, context.site.origin);
}

/**
 * `/artists/$slug`: the artist from the same list. A slug that can't be an
 * artist's is a not-found before any API call; hidden, deleted, renamed and
 * unknown artists aren't in the list, so they are the same not-found.
 */
export async function loadArtistPage(
	context: PageContext,
	slug: string,
): Promise<ArtistPageData> {
	if (!ARTIST_SLUG.test(slug)) throw notFound();
	const [studio, artists] = await Promise.all([
		loadStudio(context),
		loadArtists(context),
	]);
	const data = artistPageData(studio, artists, slug, context.site.origin);
	if (!data) throw notFound();
	return data;
}

async function loadArtists({ queryClient, tattoo }: ApiContext): Promise<Artist[]> {
	try {
		return await queryClient.query({
			...tattooQueries.artists(tattoo.client),
			staleTime: "static",
		});
	} catch (error) {
		return rethrowAsNotFound(error);
	}
}

function channelsFor(studio: StudioView, inquiry: InquiryState): InquiryChannels {
	return inquiryChannels(
		studio,
		inquiry.kind === "open" ? inquiry.preferredArtist : null,
	);
}
```

```tsx
import { createRouter as createTanStackRouter } from "@tanstack/react-router";
import { setupRouterSsrQueryIntegration } from "@tanstack/react-router-ssr-query";
import { type DehydratedApi, dehydrateApi, hydrateApi } from "./api/context.ts";
import { createRequestApi } from "./api/request.ts";
import { routeTree } from "./routeTree.gen";
import { type DehydratedSite, dehydrateSite, hydrateSite } from "./site/context.ts";
import { createRequestSite } from "./site/request.ts";
import { NotFoundPage, RouteErrorPage } from "./studio/components/pages.tsx";

type Dehydrated = DehydratedApi & DehydratedSite;

/**
 * One router per server request and one per browser tab (Start calls this
 * for each), each with its own QueryClient and SDK client (TAT-23,
 * src/api/) and the page's own origin (TAT-48, src/site/). The server sends
 * the browser only the API's origin, the studio's slug and the page's
 * origin, in the router's dehydrated state.
 */
export function getRouter() {
	const api = createRequestApi();
	const site = createRequestSite();

	const router = createTanStackRouter({
		routeTree,
		context: { queryClient: api.queryClient, tattoo: api.tattoo, site },
		scrollRestoration: true,
		defaultPreload: "intent",
		defaultPreloadStaleTime: 0,
		defaultNotFoundComponent: NotFoundPage,
		defaultErrorComponent: RouteErrorPage,
		dehydrate: (): Dehydrated => ({ ...dehydrateApi(api), ...dehydrateSite(site) }),
		hydrate: (dehydrated: Dehydrated) => {
			hydrateApi(api, dehydrated);
			hydrateSite(site, dehydrated);
		},
	});

	// Wraps the router in the QueryClientProvider and streams the cache.
	setupRouterSsrQueryIntegration({ router, queryClient: api.queryClient });

	return router;
}

declare module "@tanstack/react-router" {
	interface Register {
		router: ReturnType<typeof getRouter>;
	}
}
```

# Slugs: validation, reserved names, old-slug aliases (TAT-24) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Tenant slugs become strict lowercase DNS labels with a reserved list. Every slug a tenant has held lives in a platform-wide registry that no other tenant can take from. Owners can change the slug in Settings, and old dashboard links redirect to the current slug.

**Architecture:** A pure module (`#modules/tenancy/slugs`) holds the rules (`checkSlug`, `RESERVED_SLUGS`, `MAX_TENANT_SLUGS`, `withTenantSlug`). The validators wrap it in a Vine rule. One migration adds `tenant_slugs`: a unique slug per row, RLS with SELECT and INSERT policies only, a format CHECK, and a trigger on `tenants` that registers every new slug. `TenancyService.changeSlug` renames, and `currentSlugFor` resolves an old slug for a member. The tenant middleware sends GET and HEAD on an old slug to the current one with a 302. A "Web address" form on the studio profile page posts to `PUT /t/:tenant/settings/slug`.

**Tech Stack:** AdonisJS 7, Lucid (PostgreSQL 18, RLS), VineJS, Inertia + React, shadcn/ui, Tailwind v4, Japa.

**Spec:** `docs/superpowers/specs/2026-10-17-tat-24-slugs-design.md`. Read its Rulings first: they decide every open question.

```ts
/** The platform's own name; refused anywhere in a slug, hyphens ignored. */
const BRAND = "tattoodrip";

/** What may be typed: ASCII letters in any case, digits and hyphens. */
const TYPED = /^[A-Za-z0-9-]+$/;

/** A DNS label: 3 to 63 characters, no hyphen at either end. */
const SHAPE = /^[a-z0-9][a-z0-9-]{1,61}[a-z0-9]$/;

export type SlugProblem = "format" | "reserved";

export type SlugCheck =
	| { ok: true; slug: string }
	| { ok: false; problem: SlugProblem };

/**
 * Checks a slug a person chose and gives it in its stored form. Anything but
 * ASCII letters, digits and hyphens is refused before lowercasing, because
 * Unicode case mapping turns lookalikes into letters (the Kelvin sign
 * U+212A becomes "k"). No two hyphens in a row: that also refuses `xn--`
 * labels, which browsers show as Unicode. At least one letter, so a slug is
 * never all digits. Does not trim: the validator does.
 */
export function checkSlug(input: string): SlugCheck {
	if (!TYPED.test(input)) return { ok: false, problem: "format" };
	const slug = input.toLowerCase();
	if (!SHAPE.test(slug) || slug.includes("--") || !/[a-z]/.test(slug)) {
		return { ok: false, problem: "format" };
	}
	if (RESERVED_SLUGS.has(slug) || slug.replaceAll("-", "").includes(BRAND)) {
		return { ok: false, problem: "reserved" };
	}
	return { ok: true, slug };
}
```

```ts
const SLUG_MESSAGES = {
	format:
		"Use 3 to 63 lowercase letters, numbers and single hyphens, with at least one letter. Don't start or end with a hyphen.",
	reserved: "This address is reserved. Try another one.",
} as const;

/**
 * A slug a person chose (TAT-24): checkSlug decides, and the field keeps
 * the stored form. Taken slugs, every old slug of every tenant included,
 * are rejected by the database (tenant_slugs) and surfaced by
 * TenancyService as SlugTakenError: a query here would only see the user's
 * own tenants under row-level security.
 */
const tenantSlug = vine.createRule(
	(value: unknown, _, field: FieldContext) => {
		if (typeof value !== "string") return;
		const checked = checkSlug(value);
		if (!checked.ok) {
			field.report(
				SLUG_MESSAGES[checked.problem],
				`slug_${checked.problem}`,
				field,
			);
			return;
		}
		field.mutate(checked.slug, field);
	},
);

export const createTenantValidator = vine.create({
	name: vine.string().trim().minLength(2).maxLength(120),
	type: vine.enum(TENANT_TYPES),
	slug: vine.string().trim().use(tenantSlug()),
	timezone: vine.string().trim().use(ianaTimezone()),
});

/**
 * The page calls the slug "your address", so the errors do too.

);
```

```ts
const TENANT = "(select app_tenant_id())";
const USER = "(select app_user_id())";

/**
 * Every slug a tenant has held, its current one included (TAT-24). The
 * unique slug is one namespace for the whole platform: once a tenant has
 * used a slug, no other tenant can ever get it, so an old link never lands
 * on someone else's studio. The index enforces that without any
 * cross-tenant read.
 *
 * The tenants_register_slug trigger writes the rows whenever a tenant is
 * created or its slug changes; app code never does. RLS has SELECT and
 * INSERT policies only, so the app role can neither change nor delete a
 * row. Releasing a slug is a deliberate operations step as the owner role.
 * The key to tenants is NO ACTION, so a hard purge of a tenant has to
 * release its slugs first.
 *
 * The CHECK is the shape slugs have always had, so every existing slug is
 * copied as it is; the stricter rules for new slugs live in
 * #modules/tenancy/slugs. The (tenant_id, id) unique index also serves
 * lookups by tenant_id.
 */
export default class extends BaseSchema {
	protected tableName = "tenant_slugs";

	async up() {
		this.schema.createTable(this.tableName, (table) => {
			table.uuid("id").primary().defaultTo(this.raw("uuidv7()"));
			table
				.uuid("tenant_id")
				.notNullable()
				.references("id")
				.inTable("tenants");
			table.string("slug", 63).notNullable().unique();
			table
				.timestamp("created_at", { useTz: true })
				.notNullable()
				.defaultTo(this.now());
			table.unique(["tenant_id", "id"]);
		});

		const statements = [
			"alter table tenant_slugs add constraint tenant_slugs_slug_format check (slug ~ '^[a-z0-9][a-z0-9-]{1,61}[a-z0-9]$')",

begin
	if not exists (
		select 1 from public.tenant_slugs s where s.slug = new.slug and s.tenant_id = new.id
	) then

end
$$`,
```

```ts
	/**
	 * Moves the tenant to `slug`, which has passed checkSlug (the validators
	 * check it). The database keeps every slug the tenant has held as its own
	 * (tenant_slugs), so old links keep resolving and no other tenant can
	 * take them; switching back to an old slug reserves nothing new. Throws
	 * SlugTakenError when another tenant holds the slug, now or before, and
	 * SlugLimitError when the tenant already holds MAX_TENANT_SLUGS and
	 * `slug` is not one of them. Call it in the tenant's context. Returns the
	 * stored tenant; the caller's copy is not updated.
	 */
	async changeSlug(tenant: Tenant, slug: string): Promise<Tenant> {
		try {
			return await withTransaction(async (trx) => {
				// The lock makes two renames of one tenant run one after the
				// other, so both can't pass the limit; the comparison uses the
				// stored slug, not the caller's copy.
				const live = await Tenant.query({ client: trx })
					.where("id", tenant.id)
					.forUpdate()
					.firstOrFail();
				if (live.slug === slug) return live;

				const held = await TenantSlug.query({ client: trx })
					.where("tenant_id", tenant.id)
					.select("slug");
				const ownOldSlug = held.some((row) => row.slug === slug);
				if (!ownOldSlug && held.length >= MAX_TENANT_SLUGS) {
					throw new SlugLimitError(tenant.id);
				}

				// The tenants_register_slug trigger registers a new slug.
				live.slug = slug;
				await live.save();
				return live;
			});
		} catch (error) {
			if (isSlugTaken(error)) throw new SlugTakenError(slug);
			throw error;
		}
	}

	/**
	 * The current slug of the user's live tenant that once had this exact
	 * slug (an old link), or null. Null also covers its current slug
	 * (nothing to redirect), an unknown slug, another tenant's old slug, a
	 * removed member and a deleted tenant, as in membershipFor.
	 */
	async currentSlugFor(user: User, slug: string): Promise<string | null> {
		const membership = await TenantMembership.query()
			.where("user_id", user.id)
			// The soft-delete hooks don't reach this subquery, so filter here.
			.whereHas("tenant", (tenant) =>
				tenant
					.whereNull("tenants.deleted_at")
					.whereHas("slugs", (slugs) => slugs.where("tenant_slugs.slug", slug)),
			)
			.preload("tenant")
			.first();

		const current = membership?.tenant.slug;
		return current && current !== slug ? current : null;
	}
```

```ts
/** `/t/` and the tenant segment at the start of a dashboard URL. */
const TENANT_PREFIX = /^\/t\/[^/?#]+/;

/**
 * The same dashboard URL under another slug: `/t/<old>/rest?query` becomes
 * `/t/<slug>/rest?query`. Throws for anything else, so a redirect built

export function withTenantSlug(url: string, slug: string): string {
	if (!TENANT_PREFIX.test(url)) {

	}
	return url.replace(TENANT_PREFIX, () => `/t/${slug}`);
}
```

```ts
import { inject } from "@adonisjs/core";
import type { HttpContext } from "@adonisjs/core/http";
import { errors } from "@vinejs/vine";
import { tenantContext } from "#middleware/tenant_middleware";
import { SlugLimitError, SlugTakenError } from "#modules/tenancy/errors";
// biome-ignore lint/style/useImportType: @inject() reads the class at runtime (decorator metadata); a type-only import breaks injection.
import TenancyService from "#modules/tenancy/services/tenancy_service";
import { MAX_TENANT_SLUGS } from "#modules/tenancy/slugs";
import { changeSlugValidator } from "#modules/tenancy/validators/tenant";

function slugError(message: string) {
	return new errors.E_VALIDATION_ERROR([
		{ field: "slug", message, rule: "slug" },
	]);
}

/**
 * The studio's web address (TAT-24), changed by its owners on the profile
 * page. Old addresses keep redirecting (the tenant middleware).
 */
@inject()
export default class StudioSlugController {
	constructor(private tenancy: TenancyService) {}

	async update(ctx: HttpContext) {
		const { tenant } = tenantContext(ctx);
		const { slug } = await ctx.request.validateUsing(changeSlugValidator);
		const previous = tenant.slug;

		let changed: typeof tenant;
		try {
			changed = await this.tenancy.changeSlug(tenant, slug);
		} catch (error) {
			if (error instanceof SlugTakenError) {
				throw slugError("This address is already taken. Try another one.");
			}
			if (error instanceof SlugLimitError) {
				throw slugError(
					`Your studio has used ${MAX_TENANT_SLUGS} addresses, the most allowed. You can still switch back to an earlier one.`,
				);
			}
			throw error;
		}

		if (changed.slug !== previous) {
			ctx.session.flash(
				"success",
				"Address changed. Links to the old address still work.",
			);
		}
		return ctx.response
			.redirect()
			.toRoute("tenant.settings.profile", { tenant: changed.slug });
	}
}
```
